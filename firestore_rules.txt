rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ▼ 管理者（あなた）のGmailアドレス。必要なら書き換えてください
    function ownerEmail() { return 'maeken.ms@gmail.com'; }

    function signedIn() {
      return request.auth != null && request.auth.token.email != null
        && request.auth.token.email_verified == true;
    }
    function myEmail() { return request.auth.token.email.lower(); }
    function isOwner() { return signedIn() && myEmail() == ownerEmail(); }
    function isScorer() {
      return signedIn() && (isOwner()
        || exists(/databases/$(database)/documents/roles/$(myEmail())));
    }

    // 保護者を含め誰でも閲覧OK
    match /games/{id} {
      allow read: if true;
      allow create, update: if isScorer();   // 速報入力：許可された人
      allow delete: if isScorer();           // 試合の削除：記録係と管理者
    }
    match /meta/{doc} {
      allow read: if true;
      allow write: if isScorer();
    }
    match /team/{doc} {
      allow read: if true;
      allow write: if isOwner();             // 名簿：管理者のみ
    }
    // 入力を許可した人の一覧：管理者だけが編集できる
    match /roles/{email} {
      allow get: if signedIn() && (myEmail() == email || isOwner());
      allow list, write: if isOwner();
    }
  }
}
